! Base Router Configuration Template
! Applies to: HQ-RTR, BR-RTR
! ============================================================

enable
configure terminal

! --- Hostname ---
hostname {{HOSTNAME}}

! --- Security Hardening ---
service password-encryption
no ip domain-lookup
enable secret {{ENABLE_SECRET}}
banner motd #
***********************************************
*  NOTICE: Unauthorized access is prohibited  *
*  All activity is monitored and logged       *
***********************************************
#

! --- Console Access ---
line console 0
 logging synchronous
 exec-timeout 10 0
 password {{CONSOLE_PASSWORD}}
 login

! --- SSH Configuration ---
ip domain-name lab.local
crypto key generate rsa modulus 2048
ip ssh version 2
username admin privilege 15 secret {{ADMIN_SECRET}}

line vty 0 4
 transport input ssh
 login local
 exec-timeout 10 0

! --- Logging ---
logging buffered 16384

end
write memory
